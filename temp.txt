Inizializza SparkSession
DataFrame_Sorgente = LeggiParquet(input_path)
CreaVistaTemporanea(DataFrame_Sorgente, "flights")
DataFrame_Risultato = EseguiSparkSQL("""
    SELEZIONA origin, month,
               CONTA E CALCOLA_MEDIA(dep_delay, arr_delay) per Fascia Bassa (<15), Media (15-60) e Alta (>60),
               MAX_BY(cancellation_code, condizionato su cancelled == 1) AS causa_principale
    DA flights
    RAGGRUPPA PER origin, month
    ORDINA PER origin, month
""")
DataFrame_Risultato.RiduciAUnicaPartizione()
ScriviCSV(DataFrame_Risultato, output_path, Delimitatore=Tab, Header=Vero)
Termina SparkSession
